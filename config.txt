# QUICK START: set your notes directory. Use absolute paths.
# --sys-watch-paths "/home/user/Notes" "/home/user/Projects/wiki"


# --- runtime/system ---

# Common desktop modules. The cmd module is intentionally excluded.
# --sys-modules "alias" "workspace" "banner" "renamer" "linker" "formatter" "archive" "sys" "git"

# Override module order. Lower number runs earlier.
# --sys-modules-priority alias=0 workspace=1 sys=2 banner=10 renamer=20 linker=22 formatter=23 archive=25 git=50

# Ignore generated/internal trees when modules should not run there.
# --sys-ignore-paths "/home/user/Notes/.archive"

# Disable opened events if editors, Git, indexers, or file managers create too
# much noise. Oneshot synthetic opened events still work independently.
# --sys-disable-opened-events

# If opened events stay enabled, keep a cooldown so repeated reads are collapsed.
# --sys-opened-event-cooldown-seconds 60

# Keep daemon logs quiet by default. Use info to see Lucy event decisions.
# Debug is for low-level investigation and may include noisy library logs.
# --sys-log-level warning

# Disable notifications completely.
# Failed notification delivery attempts are logged as notification.failed.
# --sys-notification-provider disable


# --- alias ---

# Alias creates shortcuts for module args.
# --alias-rule "b=--banner {args}" "todo=--formatter-todo" "rn=--rename {args}"


# --- linker ---

# Keep markdown links valid when a note is moved or renamed.
# --linker-auto-update-md-links

# Cleanup for --linker-root: if a note used to create a root symlink and later
# no longer has --linker-root, remove the old symlink.
# --linker-auto-clean-root-links

# Ignore files or link names by basename, repo-relative path, or absolute path.
# --linker-ignore "AGENTS.md"


# --- renamer ---

# On create, rename any one-letter scratch file to a dated filename. Useful when you need a quick note but do not know how to name it yet.
# --rename-auto
# Set the output extension. Default: Markdown (`md`).
# --rename-auto-format md


# --- archive ---

# Text mode appends old content to an archive file under a date header.
# File mode creates a new dated archive file.
# --archive-default-mode text

# Archive one active scratch note into one archive file after idle hours.
# --archive-auto-pair "/home/user/Notes/now.md" "/home/user/Notes/past.md" 12 text

# Archive a file beside itself. If .archive/ exists nearby, Lucy uses it.
# --archive-auto-local "/home/user/Notes/inbox.md" 24 text

# Use one global archive destination for configured sources.
# --archive-global-dest-path "/home/user/Notes/archive.md"
# --archive-auto-global "/home/user/Notes/inbox.md" 24 text

# Tune archive date headers.
# --archive-date-prefix "--- "
# --archive-date-suffix ""


# --- plasma_widget ---

# Enable plasma_widget in --sys-modules when using these paths.

# --plasma-markdown-note-path "/home/user/Notes/todo.md"

# Optional bold-only mirror widget.
# --plasma-bold-widget-path "/home/user/.local/share/plasma_notes/<bold-widget-id>"

# Experimental: render checkboxes as CSS markers instead of plain text.
# --plasma-css-style


# --- git ---

# Enable git in --sys-modules and make sure the git binary works in PATH.

# Disable Git reaction to opened events. Useful when editor/indexer reads should
# not trigger sync, while creates/modifies/moves/deletes still do.
# --git-sync-on-opened-disable

# Merge conflict policy. Default is union: keep both local and remote lines and
# remove conflict markers. Other modes: none (leave unresolved), ours (keep
# local), theirs (keep remote), markers (commit conflict markers).
# --git-merge-autoresolve union
